function PriceCell({ price, animate }: { price: number, animate?: boolean }) {
  const [blink, setBlink] = useState('');
  const prevPrice = useRef(price);

  useEffect(() => {
    if (!animate) return;
    if (price > prevPrice.current) {
      setBlink('');
      setTimeout(() => setBlink('flash-green'), 10);
    } else if (price < prevPrice.current) {
      setBlink('');
      setTimeout(() => setBlink('flash-red'), 10);
    }
    prevPrice.current = price;
  }, [price, animate]);

  return (
    <td className={cn("px-2 py-1.5 text-right font-mono font-medium text-text-3 transition-colors", blink)}>
      ${price.toFixed(2)}
    </td>
  );
}

function BlinkingPriceText({ price, className, formattedText }: { price: number, className?: string, formattedText?: string }) {
  const [blink, setBlink] = useState('');
  const prevPrice = useRef(price);

  useEffect(() => {
    if (price > prevPrice.current) {
      setBlink('');
      setTimeout(() => setBlink('flash-green'), 10);
    } else if (price < prevPrice.current) {
      setBlink('');
      setTimeout(() => setBlink('flash-red'), 10);
    }
    prevPrice.current = price;
  }, [price]);

  return (
    <span className={cn("transition-colors rounded px-0.5", blink, className)}>
      {formattedText || price.toFixed(2)}
    </span>
  );
}
